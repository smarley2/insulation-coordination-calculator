- [ ] **Step 3: Implement `crosscheck.py`**

```python
"""Compare one standard's grid against another already-reviewed grid in the same draft.

Similar numbers are not proof of equivalence, so a comparison either proves every
mapped cell equal and yields a mapping, or blocks with a review item naming each
divergence. Nothing here re-reads a PDF: the comparison is a pure function of the
draft, so it is reproducible for a given draft.
"""

from __future__ import annotations

from typing import TYPE_CHECKING

from pydantic import model_validator

from insulation_coordination.domain.project import FrozenModel
from insulation_coordination.domain.rules import (
    CompatibilityMapping,
    Identifier,
    SourceReference,
)
from insulation_coordination.rules.importer.extract import ImportReviewItem

if TYPE_CHECKING:  # pragma: no cover
    from insulation_coordination.rules.importer.extract import ImportedRuleDraft, RawGrid


class CrossStandardCheckSpec(FrozenModel):
    id: Identifier
    source_rule_id: Identifier
    target_rule_id: Identifier
    family: Identifier
    #: (source cell id, target cell id) pairs, where a cell id is
    #: "<logical_row>/<logical_column>" as the raw grid records them.
    cell_map: tuple[tuple[Identifier, Identifier], ...]
    #: Every data cell the source grid contains. Declared separately so a partial map
    #: cannot silently compare a subset and claim equivalence.
    source_data_cell_ids: tuple[Identifier, ...]
    source: SourceReference
    notes: str = ""

    @model_validator(mode="after")
    def _map_covers_every_source_data_cell(self) -> CrossStandardCheckSpec:
        mapped = {source_id for source_id, _target_id in self.cell_map}
        if mapped != set(self.source_data_cell_ids):
            raise ValueError("cell map must cover every source data cell exactly once")
        if len(self.cell_map) != len(mapped):
            raise ValueError("cell map must not repeat a source cell")
        return self


def _cells(grid: RawGrid) -> dict[str, str]:
    return {
        f"{cell.logical_row}/{cell.logical_column}": cell.raw_text
        for cell in grid.cells
        if cell.role == "data"
    }


def compare_across_standards(
    draft: ImportedRuleDraft,
    spec: CrossStandardCheckSpec,
) -> tuple[CompatibilityMapping | None, tuple[ImportReviewItem, ...]]:
    """Prove or refute equivalence between two grids already inside the draft."""

    grids = {grid.id: grid for grid in draft.raw_grids}
    missing_grid = tuple(
        rule_id for rule_id in (spec.source_rule_id, spec.target_rule_id) if rule_id not in grids
    )
    if missing_grid:
        return None, (
            ImportReviewItem(
                code="CROSS_STANDARD_GRID_MISSING",
                semantic_id=spec.id,
                kind="mapping",
                source=spec.source,
                expected_contract=f"draft is missing grid {', '.join(missing_grid)}",
            ),
        )
    source_cells = _cells(grids[spec.source_rule_id])
    target_cells = _cells(grids[spec.target_rule_id])
    items: list[ImportReviewItem] = []
    for source_id, target_id in spec.cell_map:
        if source_id not in source_cells:
            items.append(
                ImportReviewItem(
                    code="CROSS_STANDARD_SOURCE_MISSING",
                    semantic_id=spec.id,
                    kind="mapping",
                    source=spec.source,
                    expected_contract=f"source cell {source_id} is absent from the grid",
                )
            )
            continue
        if target_id not in target_cells:
            items.append(
                ImportReviewItem(
                    code="CROSS_STANDARD_TARGET_MISSING",
                    semantic_id=spec.id,
                    kind="mapping",
                    source=spec.source,
                    expected_contract=f"target cell {target_id} is absent from the grid",
                )
            )
            continue
        if source_cells[source_id] != target_cells[target_id]:
            items.append(
                ImportReviewItem(
                    code="CROSS_STANDARD_VALUE_DIVERGENCE",
                    semantic_id=spec.id,
                    kind="mapping",
                    source=spec.source,
                    expected_contract=(
                        f"source cell {source_id} and target cell {target_id} differ; "
                        "a maintainer decides whether the standards are equivalent here"
                    ),
                )
            )
    if items:
        return None, tuple(items)
    mapping = CompatibilityMapping(
        id=spec.id,
        source_rule_id=spec.source_rule_id,
        target_rule_id=spec.target_rule_id,
        approved=False,
        source=spec.source,
        notes=spec.notes,
    )
    return mapping, ()


__all__ = ["CrossStandardCheckSpec", "compare_across_standards"]
```
